# PBR override for vp_skinned_lit_fog (skinned head/mask/items, no envmap).

# texcoord[4]/[5] so the paired FP overrides (fp_model_headgear_legacy etc.)
# can sample the directional-sun shadow map. Stock per-vertex lighting kept.
PARAM mvMatrix[4] = { state.matrix.modelview[0] };

ATTRIB vWght = vertex.attrib[1];

TEMP vNormalWorld;
ALIAS vFinalColour = vReg9;
MUL   vReg0, vertex.attrib[4], program.env[16].zzzz;
ARL   A0.x, vReg0.x;
DP4   vReg1.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg1.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg1.z, vertex.position, boneArray[A0.x + 2];

DP4   vReg2.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg2.z, vertex.position, boneArray[A0.x + 2];

DP3   vReg6.y, vertex.normal, boneArray[A0.x + 1];

DP4   vReg3.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg3.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg3.z, vertex.position, boneArray[A0.x + 2];

DP3   vReg7.y, vertex.normal, boneArray[A0.x + 1];
DP3   vReg7.z, vertex.normal, boneArray[A0.x + 2];

DP4   vReg4.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg4.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg4.z, vertex.position, boneArray[A0.x + 2];
DP3   vReg8.x, vertex.normal, boneArray[A0.x + 0];

MUL   vReg10, vWght.x, vReg5;
MAD   vReg10, vReg6, vWght.y, vReg10;
MAD   vReg10, vReg7, vWght.z, vReg10;

DP4   result.position.x, mvpMatrix[0], vReg9;
DP4   result.position.y, mvpMatrix[1], vReg9;

DP4   result.position.w, mvpMatrix[3], vReg9;

DP3   vNormalWorld.y, mvMatrix[1], vReg10;
DP3   vNormalWorld.z, mvMatrix[2], vReg10;

RSQ   vNormalWorld.w, vNormalWorld.w;
MUL   vNormalWorld, vNormalWorld, vNormalWorld.w;
DP4   vReg10.x, mvMatrix[0], vReg9;
DP4   vReg10.y, mvMatrix[1], vReg9;
DP4   vReg10.z, mvMatrix[2], vReg9;
MOV   vReg10.w, mvMatrix[3].w;
DP3	vReg0.x, vReg10, vReg10;
RSQ	vReg0.x, vReg0.x;
RCP	vReg0.x, vReg0.x;
