!!ARBvp1.0
# PBR-aware VP override for vp_static_bump_env.
# Pairs with fp_worldtex_bump_env / _gamma and fp_model_bump_env_spec on static
# bumpmapped surfaces (M4-78 doors, reflective armor props).
#
# Replaces stock tangent-basis cube outputs with PBR layout:
#   texcoord[0] = diffuse UV
#   texcoord[1] = cube reflect coord
#   texcoord[4] = world-space normal
#   texcoord[5] = world-space position

PARAM mvM[4]   = { state.matrix.modelview[0] };
PARAM mvpM[4]  = { state.matrix.mvp };
PARAM fogP     = state.fog.params;
TEMP  vTemp, vReg0, vReg1, vReg2, vReg3, vReg4, vReg10, vNormalWorld, t, wPos, wNorm;
ALIAS vFinalColour = vTemp;

# === Clip-space position ===
DP4 result.position.x, mvpM[0], vertex.position;
DP4 result.position.y, mvpM[1], vertex.position;
DP4 result.position.z, mvpM[2], vertex.position;
DP4 result.position.w, mvpM[3], vertex.position;

# === Eye-space position ===
DP4 vReg10.x, mvM[0], vertex.position;
DP4 vReg10.y, mvM[1], vertex.position;
DP4 vReg10.z, mvM[2], vertex.position;
DP4 vReg10.w, mvM[3], vertex.position;

# === Eye-space normal ===
DP3 vNormalWorld.x, mvM[0], vertex.normal;
DP3 vNormalWorld.y, mvM[1], vertex.normal;
DP3 vNormalWorld.z, mvM[2], vertex.normal;

# === Fog ===
DP3 vReg0.x, vReg10, vReg10;
RSQ vReg0.x, vReg0.x;
RCP vReg0.x, vReg0.x;
MOV result.fogcoord, vReg0.x;
ADD vReg1.x, -fogP.y, vReg0.x;
SLT t.x, 0.0, state.fog.color;
MUL t.y, vReg1.x, fogP.w;
MUL result.color.secondary.x, t.y, t.x;

# === Cube reflect coord (eye→world) ===
DP3 vReg0.x, vNormalWorld, -vReg10;
MUL vReg1, vReg0.x, vNormalWorld;
MUL vReg2, vReg1, program.env[16].y;
ADD vReg3, vReg10, vReg2;
DP3 vReg3.w, vReg3, vReg3;
RSQ vReg3.w, vReg3.w;
MUL vReg3, vReg3, vReg3.w;
DP3 vReg4.x, program.env[92], vReg3;
DP3 vReg4.y, program.env[91], vReg3;
DP3 vReg4.z, program.env[90], vReg3;
MOV vReg4.w, program.env[16].w;

# === Stock texcoords ===
MOV result.texcoord[0], vertex.texcoord[0];
MOV result.texcoord[1], vReg4;

# === World-space transform of position and normal ===
DP4 wPos.x, vReg10, program.env[92];
DP4 wPos.y, vReg10, program.env[91];
DP4 wPos.z, vReg10, program.env[90];
DP4 wPos.w, vReg10, program.env[89];
DP3 wNorm.x, vNormalWorld, program.env[92];
DP3 wNorm.y, vNormalWorld, program.env[91];
DP3 wNorm.z, vNormalWorld, program.env[90];
MOV wNorm.w, 0.0;

# === World outputs for FP per-fragment Lambert/spec ===
MOV result.texcoord[4], wNorm;
MOV result.texcoord[5], wPos;

# === Stock per-vertex lighting ===
MOV result.color, program.env[15].wwwz;
MOV vFinalColour, program.env[89].xxxw;
MOV vReg10, wPos;
MOV vNormalWorld, wNorm;

# Light 0
ADD vReg0, -vReg10, program.env[87];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[83];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[86];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[85];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

# Light 1
ADD vReg0, -vReg10, program.env[82];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[78];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[81];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[80];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

# Light 2
ADD vReg0, -vReg10, program.env[77];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[73];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[76];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[75];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

ADD vFinalColour, vFinalColour, program.env[93];
MOV vFinalColour.w, program.env[94];
MOV result.color, vFinalColour;
END
