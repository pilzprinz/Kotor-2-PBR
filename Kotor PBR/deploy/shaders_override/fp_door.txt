# PBR override of fp_door (diffuse x light, then cube-reflection LRP by diffuse alpha).
# Adds directional sun diffuse + sun-shadow receive on the diffuse part (before the
# cube blend). Reads world normal texcoord[4] / world pos texcoord[5]; guarded so a
# VP that doesn't supply them makes the sun/shadow a no-op.

ATTRIB T0 = fragment.texcoord[0];
ATTRIB T1 = fragment.texcoord[1];
ATTRIB v0 = fragment.color.primary;
ATTRIB N4 = fragment.texcoord[4];
ATTRIB P5 = fragment.texcoord[5];

PARAM c1   = program.env[1];
PARAM tnD  = program.env[27];
PARAM tnE  = program.env[28];
PARAM tnF  = program.env[29];
PARAM tnG  = program.env[30];
PARAM lightK[4] = { program.env[100..103] };  # world path: last frame's K (door pairs vp_static_env_fog, same as fp_worldtex_env_fog)
PARAM pcfStep   = program.env[32];

TEMP temp1, r0, r1, r2, alb, N, tmp, sun, shClip, shUV, shUV2, shZ, shDepth, shadowed;

TEX r0, T0, texture[0], 2D;
TEX r1, T1, texture[1], CUBE;
MOV alb, r0;                     # raw albedo for sun term
MAD temp1, c1, 2, -1.0;
ADD r2.rgb, v0, temp1;
MUL r0.rgb, r0, r2;              # stock: diffuse x (vertexlight + bias)

# Geometric world normal (MAX guards RSQ vs a zero normal)
DP3 N.w, N4, N4;
MAX N.w, N.w, 0.0001;
RSQ N.w, N.w;
MUL N.xyz, N4, N.w;

# === Shadow sample (world path: TMU6 + lightK env[100..103]) ===
# @include shadow_receive

# Sun diffuse on the lit diffuse, shadowed
DP3 tmp.x, N, tnD;
MAX tmp.x, tmp.x, 0.0;
MUL tmp.x, tmp.x, tnE.x;
MUL tmp.x, tmp.x, shadowed.x;
MUL sun.rgb, alb, tnE.yzwx;
MAD r0.rgb, sun, tmp.x, r0;

# Sun-shadow global contrast (gated by N4.sun)
DP3 tmp.x, N4, tnD;
MAX tmp.x, tmp.x, 0.0;
MUL tmp.x, tmp.x, 4.0;
MIN tmp.x, tmp.x, 1.0;
SUB tmp.y, 1.0, shClip.x;
MUL tmp.y, tmp.y, tmp.x;
MUL tmp.y, tmp.y, tnG.x;
SUB tmp.y, 1.0, tmp.y;
MAX tmp.y, tmp.y, tnF.z;
MUL r0.rgb, r0, tmp.y;

# Stock cube reflection blend + alpha (unchanged)
LRP r0.rgb, r0.a, r0, r1;
MOV r0.a, c1.a;
MOV result.color, r0;
END
