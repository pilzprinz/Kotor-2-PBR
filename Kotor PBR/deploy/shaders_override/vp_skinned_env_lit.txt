!!ARBvp1.0
# PBR-aware VP override for vp_skinned_env_lit.
# Pairs with fp_model_env_fog (skinned characters/objects + envmap).
# Adds world-space normal/position outputs to texcoord[4]/[5] for FP per-fragment
# Lambert + specular. Preserves stock outputs (clip pos, cube reflect, fog, vertex lit).

PARAM mvMatrix[4] = { state.matrix.modelview[0] };
PARAM mvpMatrix[4] = { state.matrix.mvp };
PARAM boneArray[51] = { program.env[18..68] };
PARAM fogParams = state.fog.params;
ATTRIB vTex0   = vertex.texcoord[0];
ATTRIB vWeight = vertex.attrib[1];
ADDRESS A0;
TEMP vReg4, vReg5, vReg6, vReg7, vReg8, vReg9, vReg10;
TEMP vReg0, vReg1, vReg2, vReg3;
TEMP vNormalWorld, wPos, wNorm;
ALIAS vFinalColour = vReg9;

# === Bone-weighted position (4 bones) ===
MUL   vReg0, vertex.attrib[4], program.env[16].zzzz;
ARL   A0.x, vReg0.x;
DP4   vReg1.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg1.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg1.z, vertex.position, boneArray[A0.x + 2];
DP3   vReg5.x, vertex.normal, boneArray[A0.x + 0];
DP3   vReg5.y, vertex.normal, boneArray[A0.x + 1];
DP3   vReg5.z, vertex.normal, boneArray[A0.x + 2];
ARL   A0.x, vReg0.y;
DP4   vReg2.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg2.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg2.z, vertex.position, boneArray[A0.x + 2];
DP3   vReg6.x, vertex.normal, boneArray[A0.x + 0];
DP3   vReg6.y, vertex.normal, boneArray[A0.x + 1];
DP3   vReg6.z, vertex.normal, boneArray[A0.x + 2];
ARL   A0.x, vReg0.z;
DP4   vReg3.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg3.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg3.z, vertex.position, boneArray[A0.x + 2];
DP3   vReg7.x, vertex.normal, boneArray[A0.x + 0];
DP3   vReg7.y, vertex.normal, boneArray[A0.x + 1];
DP3   vReg7.z, vertex.normal, boneArray[A0.x + 2];
ARL   A0.x, vReg0.w;
DP4   vReg4.x, vertex.position, boneArray[A0.x + 0];
DP4   vReg4.y, vertex.position, boneArray[A0.x + 1];
DP4   vReg4.z, vertex.position, boneArray[A0.x + 2];
DP3   vReg8.x, vertex.normal, boneArray[A0.x + 0];
DP3   vReg8.y, vertex.normal, boneArray[A0.x + 1];
DP3   vReg8.z, vertex.normal, boneArray[A0.x + 2];

# Blend
MUL   vReg9, vWeight.x, vReg1;
MAD   vReg9, vReg2, vWeight.y, vReg9;
MAD   vReg9, vReg3, vWeight.z, vReg9;
MAD   vReg9, vReg4, vWeight.w, vReg9;
MOV   vReg9.w, program.env[16].w;
MUL   vReg10, vWeight.x, vReg5;
MAD   vReg10, vReg6, vWeight.y, vReg10;
MAD   vReg10, vReg7, vWeight.z, vReg10;
MAD   vReg10, vReg8, vWeight.w, vReg10;

# === Clip-space ===
DP4   result.position.x, mvpMatrix[0], vReg9;
DP4   result.position.y, mvpMatrix[1], vReg9;
DP4   result.position.z, mvpMatrix[2], vReg9;
DP4   result.position.w, mvpMatrix[3], vReg9;

# === Eye-space normal (normalized) ===
DP3   vNormalWorld.x, mvMatrix[0], vReg10;
DP3   vNormalWorld.y, mvMatrix[1], vReg10;
DP3   vNormalWorld.z, mvMatrix[2], vReg10;
MOV   vNormalWorld.w, mvMatrix[3].w;
DP3   vNormalWorld.w, vNormalWorld, vNormalWorld;
RSQ   vNormalWorld.w, vNormalWorld.w;
MUL   vNormalWorld, vNormalWorld, vNormalWorld.w;

# === Eye-space position ===
DP4   vReg10.x, mvMatrix[0], vReg9;
DP4   vReg10.y, mvMatrix[1], vReg9;
DP4   vReg10.z, mvMatrix[2], vReg9;
MOV   vReg10.w, mvMatrix[3].w;

# === Fog ===
DP3   vReg0.x, vReg10, vReg10;
RSQ   vReg0.x, vReg0.x;
RCP   vReg0.x, vReg0.x;
MOV   result.fogcoord, vReg0.x;
ADD   vReg1.x, -fogParams.y, vReg0.x;
MUL   result.color.secondary.x, vReg1.x, fogParams.w;

# === Diffuse UV ===
MOV   result.texcoord[0], vTex0;

# === Cube reflect coord ===
DP3   vReg0.x, vNormalWorld, -vReg10;
MUL   vReg1, vReg0.x, vNormalWorld;
MUL   vReg2, vReg1, program.env[16].y;
ADD   vReg3, vReg10, vReg2;
DP3   vReg3.w, vReg3, vReg3;
RSQ   vReg3.w, vReg3.w;
MUL   vReg3, vReg3, vReg3.w;
DP3   vReg4.x, program.env[92], vReg3;
DP3   vReg4.y, program.env[91], vReg3;
DP3   vReg4.z, program.env[90], vReg3;
MOV   vReg4.w, program.env[16].w;
MOV   result.texcoord[1], vReg4;

# === World-space transform of position and normal ===
DP4 wPos.x, vReg10, program.env[92];
DP4 wPos.y, vReg10, program.env[91];
DP4 wPos.z, vReg10, program.env[90];
DP4 wPos.w, vReg10, program.env[89];
DP3 wNorm.x, vNormalWorld, program.env[92];
DP3 wNorm.y, vNormalWorld, program.env[91];
DP3 wNorm.z, vNormalWorld, program.env[90];
MOV wNorm.w, 0.0;

# === NEW: world outputs for FP per-fragment Lambert/spec ===
MOV result.texcoord[4], wNorm;
MOV result.texcoord[5], wPos;

# === Stock per-vertex lighting (kept) ===
MOV vFinalColour, program.env[89].xxxw;
MOV vReg10, wPos;
MOV vNormalWorld, wNorm;

# Light 0
ADD vReg0, -vReg10, program.env[87];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[83];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[86];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[85];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

# Light 1
ADD vReg0, -vReg10, program.env[82];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[78];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[81];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[80];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

# Light 2
ADD vReg0, -vReg10, program.env[77];
DP3 vReg0.w, vReg0, vReg0;
RSQ vReg1.w, vReg0.w;
MUL vReg2, vReg0, vReg1.w;
DST vReg3, vReg0.w, vReg1.w;
DP3 vReg3.w, vReg3, program.env[73];
RCP vReg3.w, vReg3.w;
DP3 vReg0.x, vNormalWorld, vReg2;
LIT vReg0, vReg0;
MUL vReg0, vReg0, vReg3.w;
MUL vReg1, vReg0.x, program.env[76];
MAD vFinalColour.xyzw, vReg1, program.env[95], vFinalColour;
MUL vReg1, vReg0.y, program.env[75];
MAD vFinalColour.xyzw, vReg1, program.env[94], vFinalColour;

ADD vFinalColour, vFinalColour, program.env[93];
MOV vFinalColour.w, program.env[94];
MOV result.color, vFinalColour;
END
